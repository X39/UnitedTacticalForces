```typescript
export interface FieldChromeProps {
  label: ReactNode;
  hint?: ReactNode | undefined;
  error?: string | undefined;
  required?: boolean | undefined;
}
export type TextFieldProps = FieldChromeProps &
  Omit<ComponentPropsWithRef<"input">, "className" | "children" | "required">;
export function TextField(props: TextFieldProps): JSX.Element;

export interface SelectOption { value: string; label: string; disabled?: boolean | undefined; lang?: string | undefined }
export interface SelectOptionGroup { group: string; options: readonly SelectOption[] }
export type SelectFieldProps = FieldChromeProps &
  Omit<ComponentPropsWithRef<"select">, "className" | "children" | "required"> & {
    options: readonly (SelectOption | SelectOptionGroup)[];
  };
export function SelectField(props: SelectFieldProps): JSX.Element;

export type CheckboxFieldProps = Omit<FieldChromeProps, "required"> &
  Omit<ComponentPropsWithRef<"input">, "className" | "children" | "type">;
export function CheckboxField(props: CheckboxFieldProps): JSX.Element;

export interface RadioOption { value: string; label: ReactNode; help?: ReactNode | undefined; disabled?: boolean | undefined }
export interface RadioGroupFieldProps {
  legend: ReactNode;          // may hold a heading element (legend allows heading content)
  name: string;
  value: string;
  options: readonly RadioOption[];
  onChange(value: string): void;
  reason?: ReactNode | undefined;   // visible text explaining the disabled options
  disabled?: boolean | undefined;
}
export function RadioGroupField(props: RadioGroupFieldProps): JSX.Element;
```
